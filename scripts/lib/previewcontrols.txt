type Kind = 'cards' | 'drag' | 'sort' | 'poll' | 'polltell' | 'split' | 'trend' | 'trend3' | 'order' | 'odd';
type State = 'open' | 'right' | 'wrong';
const KINDS: Kind[] = ['cards', 'drag', 'sort', 'poll', 'polltell', 'split', 'trend', 'trend3', 'order', 'odd'];
const STATES: State[] = ['open', 'right', 'wrong'];

/** The middle of a zone list's correct (or first wrong) zone. */
function zoneMid(zones: { upto: number; correct?: boolean }[], want: boolean) {
  const k = zones.findIndex((z) => Boolean(z.correct) === want);
  const from = k <= 0 ? 0 : zones[k - 1].upto;
  return { k, mid: (from + zones[k].upto) / 2 };
}

function Case({ kind, state }: { kind: Kind; state: State }) {
  const answered = state !== 'open';
  // The two tile controls take the branch tone directly rather than through the
  // accent provider, so a Case works out its own — `lessonId` belongs to the page.
  const caseTone = stageTone(branchOfLesson(`${q('branch') ?? 'ethics'}-preview-0`));
  const right = state === 'right';
  const noop = () => {};

  let prompt = '';
  let explain = '';
  let init = 0.5;
  let semInit = 0.5;
  let picked: string | null = null;
  let body: 'cards' | 'drag' | 'sort' | 'poll' | 'split' | 'trend' | 'order' | 'odd' = 'cards';

  if (kind === 'cards') {
    prompt = CARDS.prompt; explain = CARDS.explain; body = 'cards';
    const seed = seedFor(LESSON, CARDS.cards);
    const shown = swapFor(seed) ? [CARDS.cards[1], CARDS.cards[0]] : CARDS.cards;
    const rightId = shown[0].correct ? 'c0' : 'c1';
    picked = !answered ? null : right ? rightId : rightId === 'c0' ? 'c1' : 'c0';
  } else if (kind === 'drag') {
    prompt = DRAG.prompt; explain = DRAG.explain; body = 'drag';
    const z = zoneMid(DRAG.drag.zones, right || !answered);
    init = answered ? z.mid : DRAG.drag.start;
    picked = answered ? DRAG.drag.zones[z.k].id : null;
  } else if (kind === 'split') {
    prompt = SPLIT.prompt; explain = SPLIT.explain; body = 'split';
    const z = zoneMid(SPLIT.split.zones, right || !answered);
    init = answered ? Math.min(0.94, Math.max(0.06, z.mid)) : SPLIT.split.start;
    picked = answered ? SPLIT.split.zones[z.k].id : null;
  } else if (kind === 'sort') {
    prompt = SORT.prompt; explain = SORT.explain; body = 'sort';
    const bins = SORT.sort.bins;
    const a = bins.findIndex((b) => Boolean(b.correct) === (right || !answered));
    const order = orderFor(seedFor(LESSON, [{ text: SORT.sort.chip }]), bins.length);
    const d = order.indexOf(a);
    init = answered ? (d + 0.5) / bins.length : 0.5;
    semInit = answered ? a / (bins.length - 1) : 0.5;
    picked = answered ? bins[a].id : null;
  } else if (kind === 'poll' || kind === 'polltell') {
    const P = kind === 'poll' ? POLL : POLL_TELL;
    prompt = P.prompt; explain = P.explain; body = 'poll';
    const o = P.poll.options.find((x) => Boolean(x.correct) === right) ?? P.poll.options[0];
    picked = answered ? o.id : null;
  } else if (kind === 'order') {
    prompt = ORDER.prompt; explain = ORDER.explain; body = 'order';
    // An order is answered by placing EVERY tile, so the answered state is a full
    // sequence: the authored one when it is right, and the last two swapped when
    // it is not, which is the commonest way a reader gets one wrong.
    const n = ORDER.order.items.length;
    init = answered ? 1 : 0;
    semInit = answered ? 1 : 0.5;
    const asAuthored = [...ORDER.order.items.keys()];
    if (!right) { const t = asAuthored[n - 1]; asAuthored[n - 1] = asAuthored[n - 2]; asAuthored[n - 2] = t; }
    picked = answered ? asAuthored.join('>') : null;
  } else if (kind === 'odd') {
    prompt = ODD.prompt; explain = ODD.explain; body = 'odd';
    const tiles = ODD.odd.tiles;
    const t = tiles.find((x) => Boolean(x.correct) === right) ?? tiles[0];
    init = answered ? (right ? 1 : 0.35) : 0;
    semInit = answered ? tiles.indexOf(t) / (tiles.length - 1) : 0.5;
    picked = answered ? t.id : null;
  } else {
    const T = kind === 'trend' ? PLOT : PLOT3;
    prompt = T.prompt; explain = T.explain; body = 'trend';
    const s = T.plot.shapes.find((x) => Boolean(x.correct) === right) ?? T.plot.shapes[0];
    picked = answered ? s.id : null;
  }

  // A split STARTS at its opening value and is then moved, as a reader moves it. Its
  // two percentages are written by Reanimated only when the value differs from what
  // was drawn at render, so rendering at the answer would leave them at defaultValue.
  // SplitBar also puts its value back to split.start when it mounts, and a child's
  // effect runs before its parent's, so the answered position is applied here.
  const pos = useSharedValue(kind === 'split' ? SPLIT.split.start : init);
  const sem = useSharedValue(semInit);
  useEffect(() => { pos.value = init; sem.value = semInit; }, [pos, sem, init, semInit]);
  const P = kind === 'polltell' ? POLL_TELL : POLL;
  const T = kind === 'trend3' ? PLOT3 : PLOT;

  return (
    <View nativeID={`case-${kind}-${state}`} style={{ width: 390, paddingTop: 8, paddingBottom: 14, borderBottomWidth: 1, borderBottomColor: '#D8D4CA' }}>
      <Text style={{ fontSize: 10, color: '#9A968C', paddingHorizontal: 24, marginBottom: 6, letterSpacing: 1 }}>
        {`${kind.toUpperCase()} · ${state.toUpperCase()}`}
      </Text>
      <View nativeID={`ctl-${kind}-${state}`}>
        {body === 'cards' ? <ChoiceCards cards={CARDS.cards} picked={picked} onPick={noop} seed={seedFor(LESSON, CARDS.cards)} /> : null}
        {body === 'drag' ? <DragScale drag={DRAG.drag} picked={picked} onPick={noop} pos={pos} /> : null}
        {body === 'split' ? <SplitBar split={SPLIT.split} picked={picked} onPick={noop} pos={pos} /> : null}
        {body === 'sort' ? <SortBins sort={SORT.sort} picked={picked} onPick={noop} pos={pos} sem={sem} seed={seedFor(LESSON, [{ text: SORT.sort.chip }])} /> : null}
        {body === 'poll' ? <PollBallot poll={P.poll} picked={picked} onPick={noop} pos={pos} sem={sem} seed={seedFor(LESSON, [{ text: P.poll.options[0].reads }])} /> : null}
        {body === 'trend' ? <TrendPick plot={T.plot} picked={picked} onPick={noop} pos={pos} sem={sem} seed={seedFor(LESSON, [{ text: T.plot.shapes[0].reads }])} /> : null}
        {body === 'order' ? <OrderTiles order={ORDER.order} picked={picked} onPick={noop} pos={pos} sem={sem} seed={seedFor(LESSON, [{ text: ORDER.order.items[0].reads }])} tone={caseTone} /> : null}
        {body === 'odd' ? <OddOneOut odd={ODD.odd} picked={picked} onPick={noop} pos={pos} sem={sem} seed={seedFor(LESSON, [{ text: ODD.odd.tiles[0].reads }])} tone={caseTone} /> : null}
      </View>
      <View nativeID={`panel-${kind}-${state}`} style={{ paddingHorizontal: 24 }}>
        <InteractPanel prompt={prompt} explain={explain} answered={answered} correct={right} inScene={false} index={1} total={2} />
      </View>
    </View>
  );
}

export default function PreviewControls() {
  const [ready, setReady] = useState(false);
  useEffect(() => {
    useUserDataStore.setState({ _hasHydrated: true } as never);
    // The root layout holds every route behind the launch animation; a preview
    // that does not release it photographs an empty page (sheet-pass learned it).
    useUIStore.setState({ launchDone: true } as never);
    setReady(true);
  }, []);
  if (!ready) return <View style={{ flex: 1 }} nativeID="controls-boot" />;

  const only = (q('only') ?? '').split(',').filter(Boolean) as Kind[];
  const kinds = only.length ? KINDS.filter((k) => only.includes(k)) : KINDS;
  const states = (q('states') ?? '').split(',').filter(Boolean) as State[];
  const list = states.length ? STATES.filter((s) => states.includes(s)) : STATES;
  // ?branch=logic strikes the sheet in that branch's colour.
  const lessonId = `${q('branch') ?? 'ethics'}-preview-0`;

  return (
    <GestureHandlerRootView style={{ flex: 1 }}>
      <QuestionAccentProvider lessonId={lessonId}>
        <View nativeID="controls-root" style={{ flex: 1, backgroundColor: '#FAFAF7', alignItems: 'flex-start' }}>
          {kinds.map((k) => list.map((s) => <Case key={`${k}-${s}`} kind={k} state={s} />))}
        </View>
      </QuestionAccentProvider>
    </GestureHandlerRootView>
  );
}
