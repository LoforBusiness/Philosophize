import { useEffect, useState } from 'react';
import { View } from 'react-native';
import StreakPanel from '@/components/gamification/StreakPanel';
import StreakCalendar from '@/components/gamification/StreakCalendar';

import StreakCelebration from '@/components/gamification/StreakCelebration';
import StreakCeremony from '@/components/gamification/StreakCeremony';

const q = (k: string) =>
  typeof window === 'undefined' ? null : new URLSearchParams(window.location.search).get(k);

/** A run of days ending today, with a couple of gaps further back. */
function seedDays(today: Date, run: number) {
  const active: string[] = [];
  const rest: string[] = [];
  const key = (d: Date) => d.toISOString().slice(0, 10);
  for (let i = 0; i < run; i++) {
    const d = new Date(today);
    d.setDate(d.getDate() - i);
    active.push(key(d));
  }
  // one rest day bridging a gap, and a scatter before the run
  for (const back of [run + 1, run + 4, run + 5, run + 8, run + 12, run + 13, run + 14]) {
    const d = new Date(today);
    d.setDate(d.getDate() - back);
    if (back === run + 1) rest.push(key(d));
    else active.push(key(d));
  }
  return { active, rest };
}

export default function PreviewStreak() {
  const [ready, setReady] = useState(false);
  // `?d=YYYY-MM-DD` OVERRIDES TODAY, and it is not a convenience. The week strip
  // is built from the real calendar, so on a Monday a 40-day streak still draws
  // one lit day and six future rings and the run rail has nothing to span —
  // correct, and useless for judging the one element the rail exists to be.
  // Noon rather than midnight: `toISOString` on a local midnight rolls back a
  // day in any positive-offset zone, which would silently shift every seeded key.
  const dq = q('d');
  const today = dq ? new Date(`${dq}T12:00:00`) : new Date();
  const todayKey = today.toISOString().slice(0, 10);
  const run = +(q('run') ?? 12);
  const { active, rest } = seedDays(today, run);

  useEffect(() => {
    const since = new Date(today);
    since.setDate(since.getDate() - 90);
    useUserDataStore.setState({
      _hasHydrated: true,
      displayName: 'Landy',
      totalXP: 1240,
      rankIndex: 6,
      streak: run,
      lastLessonDate: q('lapsed') === '1' ? '2026-01-01' : todayKey,
      activeDays: active,
      restDays: rest,
      restDaysEarned: 2,
      restDaysUsed: 1,
      joinedAt: since.getTime(),
    } as never);
    useUIStore.setState({ launchDone: true } as never);
    useSubscriptionStore.setState({
      ready: true, available: false, isPro: false, monthly: null, refresh: async () => {},
    } as never);
    setReady(true);
  }, []);

  if (!ready) return <View style={{ flex: 1 }} nativeID="streak-boot" />;

  const s = q('s') ?? 'tab';
  let body = null;
  if (s === 'tab') {
    body = <StreakTab />;
  } else if (s === 'celebrate') {
    body = (
      <View style={{ padding: 16, backgroundColor: '#FAFAF7' }}>
        <StreakCelebration
          streak={run}
          prevStreak={run - 1}
          restSpent={0}
          activeDays={active}
          restDays={rest}
          today={todayKey}
          since={null}
        />
      </View>
    );
  } else if (s === 'ceremony') {
    // THE FULL-SCREEN MOMENT. It fills the viewport itself, so unlike every
    // other surface here it gets no padded wrapper — a box round it would be
    // measuring a different screen from the one that ships.
    body = (
      <StreakCeremony
        streak={run}
        prevStreak={run - 1}
        restSpent={q('rest') === '1' ? 1 : 0}
        activeDays={active}
        restDays={rest}
        today={todayKey}
        since={null}
        moreToday={q('more') === '1'}
        onDone={() => {}}
      />
    );
  } else if (s === 'panel') {
    body = (
      <View style={{ padding: 16, gap: 16, backgroundColor: '#FAFAF7' }}>
        <View style={{ backgroundColor: '#1A1A1A', borderRadius: 14, padding: 16 }}>
          <StreakPanel
            streak={run}
            lastLessonDate={todayKey}
            restHeld={1}
            restMax={2}
            restBridging={false}
            onInk
          />
        </View>
        <View style={{ backgroundColor: '#FAFAF7', borderRadius: 14, padding: 16, borderWidth: 1, borderColor: '#E7E3DA' }}>
          <StreakPanel
            streak={run}
            lastLessonDate={todayKey}
            restHeld={1}
            restMax={2}
            restBridging={false}
          />
        </View>
      </View>
    );
  } else {
    body = (
      <View style={{ padding: 16, backgroundColor: '#FAFAF7' }}>
        <StreakCalendar
          activeDays={active}
          restDays={rest}
          today={todayKey}
          since={null}
        />
      </View>
    );
  }

  return (
    <View nativeID="streak-root" style={{ flex: 1, backgroundColor: '#FAFAF7' }}>
      {body}
    </View>
  );
}
