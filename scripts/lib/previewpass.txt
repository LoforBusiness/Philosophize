// TEMPORARY — DELETE BEFORE COMMITTING. Any file in app/ is a real route (§21).
// Written and removed by scripts/sheet-pass.mjs; check-routes.mjs blocks a build
// if one is left behind.
import { useEffect, useState } from 'react';
import { View } from 'react-native';
import PaywallContent from '@/components/shared/PaywallContent';
import PassTab from '@/app/(app)/pass';
import SettingsScreen from '@/app/(app)/settings';
import DailyLimit from '@/components/paywall/DailyLimit';
import LessonLocked from '@/components/paywall/LessonLocked';
import TrialOffer from '@/components/paywall/TrialOffer';
import PassConferred from '@/components/paywall/PassConferred';
import ProfileScreen from '@/app/(app)/profile';
import { useUserDataStore } from '@/stores/userDataStore';
import { useUIStore } from '@/stores/uiStore';
import { useSubscriptionStore } from '@/stores/subscriptionStore';
import { notifications } from '@/lib/notifications';
import { ALL_BRANCHES } from '@/data';

const q = (k: string) =>
  typeof window === 'undefined' ? null : new URLSearchParams(window.location.search).get(k);

const HOUR = 3_600_000;

// THE STORE'S MONTHLY PACKAGE, seeded the way `normalize` in lib/purchases/real.ts
// builds it. The web stub has no RevenueCat, so without this no screen could show
// the free trial at all: the trial is read off the package, never assumed.
const PKG = {
  identifier: '$rc_monthly',
  productId: 'philosophize_scholars_pass_monthly:monthly',
  priceString: '$6.99',
  price: 6.99,
  currency: 'USD',
  period: 'month',
  trial: { value: 3, unit: 'day' },
  // THE CHARGE-TODAY OPTION, seeded for the same reason the trial is: it is read
  // off the store, never assumed, so without it the door under the trial button
  // would not be drawn and this harness would photograph the screen without the
  // thing it is supposed to be measuring — and report it clean. `?base=none`
  // renders the other half of that decision, where Google named no such option.
  basePlan: { priceString: '$6.99', price: 6.99, currency: 'USD' },
};

export default function PreviewPass() {
  const [ready, setReady] = useState(false);

  useEffect(() => {
    const fresh = q('seed') === 'new';
    const kind = q('k') ?? 'replay';

    // A mid-journey reader: a few lessons in every branch, a live streak, enough
    // XP to hold a rank with an arc toward the next one.
    const byUnit: Record<string, number> = {};
    const byBranch: Record<string, number> = {};
    ALL_BRANCHES.forEach((b, bi) => {
      let n = 0;
      b.paths.forEach((u, ui) => {
        const done = fresh ? 0 : ui === 0 ? Math.min(u.lessons.length, 2 + bi) : 0;
        byUnit[u.id] = done;
        n += done;
      });
      byBranch[b.slug] = n;
    });

    // The three locked readings need the unit counts arranged to produce them.
    const branch = ALL_BRANCHES[2]; // logic
    const unit0 = branch.paths[0];
    const unit1 = branch.paths[1];
    if (kind === 'replay') byUnit[unit0.id] = 3;
    if (kind === 'unreached') byUnit[unit0.id] = 1;
    if (kind === 'ahead') { byUnit[unit0.id] = 2; byUnit[unit1.id] = 0; }

    useUserDataStore.setState({
      _hasHydrated: true,
      displayName: fresh ? 'Philosopher' : 'Landy',
      totalXP: fresh ? 0 : 1240,
      rankIndex: fresh ? 0 : 6,
      streak: fresh ? 0 : 12,
      lastLessonDate: new Date().toISOString().slice(0, 10),
      restDaysEarned: fresh ? 0 : 1,
      restDaysUsed: 0,
      lessonsByUnit: byUnit,
      lessonsByBranch: byBranch,
      savedQuotes: [],
      philosopherViews: {},
    } as never);

    // NOTIFICATIONS, AS A PHONE WOULD ANSWER. The web build has no scheduler, so
    // the trial's reminder ask renders nothing there, and it is the one piece of
    // this flow a reader on a phone always meets. `notify=ask` is a phone that has
    // not been asked; `notify=granted` one that allowed it.
    const notify = q('notify');
    if (notify === 'ask' || notify === 'granted') {
      Object.assign(notifications, {
        isSupported: () => true,
        hasPermission: async () => notify === 'granted',
        requestPermission: async () => false,
      });
    }

    // The root layout holds every route behind the launch animation, so a
    // preview that does not release it photographs an empty page with no error
    // anywhere — which is exactly what the first run of this harness did.
    useUIStore.setState({ launchDone: true } as never);

    // `refresh` IS NEUTRALISED, not just the flag set. PaywallContent calls it
    // on mount and the web stub answers "not subscribed", so a seeded `isPro`
    // was being overwritten a frame later and the subscriber state could not be
    // previewed at all — it silently showed the offer instead.
    //
    // THE FIELDS ARE SEEDED THE WAY THE STORE WOULD HOLD THEM. `isPro` is derived
    // from `entitled` and the trial, so seeding it alone would preview a state the
    // app can never be in.
    //
    //   trial=on         Google Play's trial, 60 hours from its end, will convert
    //   trial=cancelled  the same trial, cancelled: it will not convert
    //   trial=used       Google no longer offers this reader a trial
    //   trial=device     the retired on-device trial, 36 hours from its end
    const pro = q('pro') === '1';
    const trial = q('trial');
    const now = Date.now();
    const google = trial === 'on' || trial === 'cancelled';
    const NONE = { active: false, onTrial: false, expiresAt: null, willRenew: false, productId: null };
    useSubscriptionStore.setState({
      ready: true,
      available: false,
      entitled: pro || google,
      isReviewer: false,
      sub: google
        ? { active: true, onTrial: true, expiresAt: now + 60 * HOUR, willRenew: trial === 'on', productId: PKG.productId }
        : pro
          ? { active: true, onTrial: false, expiresAt: now + 20 * 24 * HOUR, willRenew: true, productId: PKG.productId }
          : NONE,
      trialEndsAt: trial === 'device' ? now + 36 * HOUR : null,
      isPro: pro || google || trial === 'device',
      monthly: (() => {
        const base = trial === 'used' ? { ...PKG, trial: null } : PKG;
        // `?base=none`: Google named no option that charges today, so the door
        // under the trial button must not be drawn at all.
        return q('base') === 'none' ? { ...base, basePlan: null } : base;
      })(),
      refresh: async () => {},
    } as never);

    setReady(true);
  }, []);

  if (!ready) return <View style={{ flex: 1 }} nativeID="pass-boot" />;

  const s = q('s') ?? 'paywall';
  const kind = q('k') ?? 'replay';
  const branch = ALL_BRANCHES[2];
  const unit0 = branch.paths[0];
  const unit1 = branch.paths[1];

  let body = null;
  if (s === 'settings') {
    // THE REAL SETTINGS SCREEN, opened on its Subscription section. It is a
    // route like the tab is, and renders outside a navigator for the same
    // reason. The section is reached by the screen's own rail rather than by a
    // prop, so the harness clicks it — see sheet-pass.mjs.
    body = <SettingsScreen />;
  } else if (s === 'profile') {
    // THE PROFILE, after the statistics tab was folded into it (2026-09-15).
    // It is here rather than in a harness of its own because this one already
    // seeds a mid-journey reader, boots a browser and measures overflow — and
    // the single thing worth asking about the new YOUR PROGRESS card is whether
    // its four counts and six reading rows survive a real layout at 320dp.
    body = <ProfileScreen />;
  } else if (s === 'tab') {
    // THE REAL TAB SCREEN, imported as a component. It is a route, so it renders
    // perfectly well outside the navigator — the only thing it needs from one is
    // `useFocusEffect`, which no-ops without a focus context, and the analytics
    // call behind it is a stub on web.
    body = <PassTab />;
  } else if (s === 'trial') {
    // THE POST-LESSON OFFER. A Modal, which react-native-web portals outside this
    // root; sheet-pass reads the whole body for that reason.
    body = <TrialOffer onAccept={async () => 'unavailable'} onDecline={() => {}} />;
  } else if (s === 'conferral') {
    // THE CEREMONY, as a trial that has just started. Also a Modal.
    body = <PassConferred kind="trial" onDone={() => {}} />;
  } else if (s === 'limit') {
    body = <DailyLimit lesson={unit0.lessons[2]} branch={branch} onExit={() => {}} />;
  } else if (s === 'locked') {
    const unit = kind === 'ahead' ? unit1 : unit0;
    const lesson = kind === 'replay' ? unit0.lessons[0]
      : kind === 'ahead' ? unit1.lessons[0]
      : unit0.lessons[3];
    body = (
      <LessonLocked
        lesson={lesson}
        branch={branch}
        unit={unit}
        gatedByPro={kind !== 'unreached'}
        onExit={() => {}}
      />
    );
  } else {
    body = <PaywallContent onClose={() => {}} source="preview" />;
  }

  return (
    <View nativeID="pass-root" style={{ flex: 1, backgroundColor: '#FAFAF7' }}>
      {body}
    </View>
  );
}
